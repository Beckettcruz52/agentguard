Developer Integration Guide

Add AegisAI to LangChain, AutoGen, CrewAI, or any custom Python agent with just 2-3 lines of code.

LangChain

Callback Handler & Memory Store

from agentguard import AgentGuardLangChainCallback, AgentGuardMemoryStore

# 1. Protect Vector Database (ChromaDB / FAISS)
safe_store = AgentGuardMemoryStore()
safe_store.add_texts(["Incoming unverified customer email..."])

# 2. Attach AegisAI Guard to Agent Executor
guard_callback = AgentGuardLangChainCallback(role="support_agent")
agent_executor.run("Answer user question", callbacks=[guard_callback])
Python SDK

Tool Decorator Guard (@protect_tool)

from agentguard import protect_tool

@protect_tool(role="devops_agent", dry_run=True)
def execute_shell_command(command: str):
    """Safely executes diagnostic commands inside isolated sandbox."""
    return subprocess.check_output(command, shell=True)

# Calling with rm -rf / or curl $AWS_SECRET automatically
# raises AgentSecurityException and stops the action!
CrewAI / AutoGen

Role-Based Access Control

from crewai import Agent
from agentguard import protect_tool

# Bind role-specific guard directly to agent capabilities
researcher = Agent(
    role="Research Assistant",
    tools=[
        protect_tool(role="guest_agent")(search_web),
        # protect_tool blocks unauthorized shell access for this role!
    ]
)
